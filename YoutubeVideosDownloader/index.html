هذا الملف هو واجهة المستخدم فقط. المتصفح لا يمكنه تنزيل فيديوهات يوتيوب مباشرة بسبب قيود CORS وسياسة يوتيوب. لذلك تحتاج إلى سيرفر صغير يعمل بـ yt-dlp على جهازك، ثم ضع رابطه هنا:

1) ثبّت المتطلبات:

pip install yt-dlp
npm install express cors

2) أنشئ ملف server.js:

const express = require('express');
const cors = require('cors');
const { execFile, spawn } = require('child_process');

const app = express();
app.use(cors());

// جلب معلومات الفيديو + الصيغ المتاحة
app.get('/info', (req, res) => {
  const url = req.query.url;
  if (!url) return res.status(400).json({ error: 'رابط مفقود' });

  execFile('yt-dlp', ['-J', '--no-warnings', url],
    { maxBuffer: 1024 * 1024 * 80 }, (err, stdout) => {
      if (err) return res.status(500).json({ error: 'تعذّر جلب الفيديو' });

      const d = JSON.parse(stdout);
      const formats = [];

      // فيديو + صوت مدمج
      d.formats
        .filter(f => f.vcodec !== 'none' && f.acodec !== 'none' && f.ext === 'mp4')
        .forEach(f => formats.push({
          id: f.format_id,
          label: `MP4 ${f.height || '?'}p`,
          type: 'video',
          size: f.filesize || f.filesize_approx || 0
        }));

      // صوت فقط
      const audio = d.formats
        .filter(f => f.vcodec === 'none' && f.acodec !== 'none')
        .sort((a, b) => (b.abr || 0) - (a.abr || 0))[0];

      if (audio) formats.push({
        id: audio.format_id,
        label: 'MP3 صوت فقط',
        type: 'audio',
        size: audio.filesize || audio.filesize_approx || 0
      });

      // إزالة التكرار وترتيب تنازلي بالجودة
      const uniq = [];
      const seen = new Set();
      for (const f of formats.reverse()) {
        if (seen.has(f.label)) continue;
        seen.add(f.label);
        uniq.push(f);
      }

      res.json({
        title: d.title,
        author: d.uploader,
        duration: d.duration,
        thumbnail: d.thumbnail,
        formats: uniq
      });
    });
});

// تنزيل الفيديو (stream مباشر)
app.get('/download', (req, res) => {
  const { url, format, type } = req.query;
  if (!url) return res.status(400).end();

  const args = type === 'audio'
    ? ['-f', format, '-x', '--audio-format', 'mp3', '-o', '-', url]
    : ['-f', format, '-o', '-', url];

  res.setHeader('Content-Type', type === 'audio' ? 'audio/mpeg' : 'video/mp4');
  res.setHeader('Content-Disposition', 'attachment');

  const proc = spawn('yt-dlp', args);
  proc.stdout.pipe(res);
  proc.stderr.on('data', () => {});
  proc.on('error', () => res.status(500).end());
});

app.listen(3000, () =>
  console.log('✅ السيرفر يعمل على http://localhost:3000'));

3) شغّل السيرفر:

node server.js

ثم ضع http://localhost:3000 في الحقل أعلاه واضغط حفظ. ✅
